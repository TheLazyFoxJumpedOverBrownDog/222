<script>
const positions = [
  {x:140, y:40}, {x:110, y:70}, {x:170, y:70},
  {x:80, y:110}, {x:200, y:110},
  {x:60, y:150}, {x:140, y:150}, {x:220, y:150},
  {x:40, y:190}, {x:100, y:190}, {x:180, y:190}, {x:240, y:190},
  {x:70, y:230}, {x:140, y:230}, {x:210, y:230},
  {x:100, y:270}, {x:170, y:270},
  {x:140, y:310}, {x:90, y:340}, {x:190, y:340},
  {x:40, y:260}, {x:240, y:260}, {x:40, y:310}, {x:240, y:310},
  {x:140, y:360}
];

let appleCount = 0;

function addApple(emoji){
  if(appleCount >= 25) return alert("Ағаштағы барлық алма орындары толды!");

  const spot = document.createElement("div");
  spot.className = "spot";
  spot.style.left = positions[appleCount].x + "px";
  spot.style.top = positions[appleCount].y + "px";
  spot.textContent = emoji;

  document.getElementById("spots").appendChild(spot);
  appleCount++;
}
</script>
